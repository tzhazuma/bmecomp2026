# src/python/bci_data_collector.py

import socket
import json
import threading
from collections import deque
from typing import Dict, Optional

class BCIDataCollector:
    """BCI数据采集器"""
    
    def __init__(self, host: str = 'localhost', port: int = 5555, buffer_size: int = 100):
        self.host = host
        self.port = port
        self.buffer_size = buffer_size
        self.attention_buffer = deque(maxlen=buffer_size)
        self.imu_buffer = deque(maxlen=buffer_size)
        self.is_running = False
        self.socket = None
        
    def start(self):
        """启动数据采集"""
        self.is_running = True
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        self.receive_thread = threading.Thread(target=self._receive_data)
        self.receive_thread.start()
        
    def stop(self):
        """停止数据采集"""
        self.is_running = False
        if self.socket:
            self.socket.close()
            
    def _receive_data(self):
        """接收数据线程"""
        while self.is_running:
            try:
                data = self.socket.recv(1024).decode('utf-8')
                if data:
                    parsed = json.loads(data)
                    self._process_data(parsed)
            except Exception as e:
                print(f"数据接收错误: {e}")
                
    def _process_data(self, data: Dict):
        """处理接收到的数据"""
        if 'attention' in data:
            self.attention_buffer.append({
                'timestamp': data.get('timestamp'),
                'value': data['attention']
            })
        if 'imu' in data:
            self.imu_buffer.append({
                'timestamp': data.get('timestamp'),
                'yaw': data['imu'].get('yaw'),
                'pitch': data['imu'].get('pitch')
            })
            
    def get_attention(self) -> Optional[float]:
        """获取最新专注力值"""
        if self.attention_buffer:
            return self.attention_buffer[-1]['value']
        return None
        
    def get_imu(self) -> Optional[Dict]:
        """获取最新IMU数据"""
        if self.imu_buffer:
            return self.imu_buffer[-1]
        return None
```

# src/python/signal_processor.py

import numpy as np
from collections import deque

class SignalProcessor:
    """信号处理器"""
    
    def __init__(self, window_size: int = 10, alpha: float = 0.7):
        self.window_size = window_size
        self.alpha = alpha
        self.attention_window = deque(maxlen=window_size)
        self.smoothed_attention = None
        
    def process_attention(self, raw_attention: float) -> float:
        """处理专注力信号"""
        # 添加到滑动窗口
        self.attention_window.append(raw_attention)
        
        # 滑动窗口滤波
        windowed = np.mean(list(self.attention_window))
        
        # 指数平滑
        if self.smoothed_attention is None:
            self.smoothed_attention = windowed
        else:
            self.smoothed_attention = self.alpha * windowed + (1 - self.alpha) * self.smoothed_attention
            
        return self.smoothed_attention
        
    def process_imu(self, yaw: float, pitch: float, deadzone: float = 5.0) -> tuple:
        """处理IMU信号"""
        # 死区处理
        if abs(yaw) < deadzone:
            yaw = 0
        if abs(pitch) < deadzone:
            pitch = 0
            
        return yaw, pitch
```

```python
# src/python/socket_server.py

import socket
import json
import threading
from typing import Callable, Dict

class SocketServer:
    """Socket服务器"""
    
    def __init__(self, host: str = '0.0.0.0', port: int = 5555):
        self.host = host
        self.port = port
        self.server_socket = None
        self.client_socket = None
        self.is_running = False
        self.on_data_received: Callable[[Dict], None] = None
        
    def start(self):
        """启动服务器"""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(1)
        self.is_running = True
        
        print(f"服务器启动在 {self.host}:{self.port}")
        
        self.accept_thread = threading.Thread(target=self._accept_connections)
        self.accept_thread.start()
        
    def stop(self):
        """停止服务器"""
        self.is_running = False
        if self.client_socket:
            self.client_socket.close()
        if self.server_socket:
            self.server_socket.close()
            
    def _accept_connections(self):
        """接受连接"""
        while self.is_running:
            try:
                client_socket, address = self.server_socket.accept()
                print(f"客户端连接: {address}")
                self.client_socket = client_socket
                
                self.receive_thread = threading.Thread(target=self._receive_data)
                self.receive_thread.start()
            except Exception as e:
                print(f"接受连接错误: {e}")
                
    def _receive_data(self):
        """接收数据"""
        while self.is_running:
            try:
                data = self.client_socket.recv(1024).decode('utf-8')
                if data:
                    parsed = json.loads(data)
                    if self.on_data_received:
                        self.on_data_received(parsed)
            except Exception as e:
                print(f"接收数据错误: {e}")
                break
                
    def send_data(self, data: Dict):
        """发送数据"""
        if self.client_socket:
            try:
                json_data = json.dumps(data)
                self.client_socket.send(json_data.encode('utf-8'))
            except Exception as e:
                print(f"发送数据错误: {e}")
```

```csharp
// src/unity/Assets/Scripts/Core/BCIManager.cs

using UnityEngine;
using System;
using System.Net.Sockets;
using System.Text;
using System.Threading;
using Newtonsoft.Json.Linq;

public class BCIManager : MonoBehaviour
{
    public static BCIManager Instance { get; private set; }
    
    [Header("连接设置")]
    public string serverHost = "127.0.0.1";
    public int serverPort = 5555;
    
    [Header("数据缓存")]
    public float currentAttention = 0f;
    public float currentYaw = 0f;
    public float currentPitch = 0f;
    
    private TcpClient client;
    private NetworkStream stream;
    private Thread receiveThread;
    private bool isConnected = false;
    
    public event Action<float> OnAttentionUpdated;
    public event Action<float, float> OnIMUUpdated;
    
    void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
        }
    }
    
    void Start()
    {
        ConnectToServer();
    }
    
    void OnDestroy()
    {
        Disconnect();
    }
    
    public void ConnectToServer()
    {
        try
        {
            client = new TcpClient(serverHost, serverPort);
            stream = client.GetStream();
            isConnected = true;
            
            receiveThread = new Thread(ReceiveData);
            receiveThread.IsBackground = true;
            receiveThread.Start();
            
            Debug.Log("已连接到BCI服务器");
        }
        catch (Exception e)
        {
            Debug.LogError($"连接BCI服务器失败: {e.Message}");
        }
    }
    
    public void Disconnect()
    {
        isConnected = false;
        if (stream != null) stream.Close();
        if (client != null) client.Close();
    }
    
    private void ReceiveData()
    {
        byte[] buffer = new byte[1024];
        
        while (isConnected)
        {
            try
            {
                int bytesRead = stream.Read(buffer, 0, buffer.Length);
                if (bytesRead > 0)
                {
                    string data = Encoding.UTF8.GetString(buffer, 0, bytesRead);
                    ProcessData(data);
                }
            }
            catch (Exception e)
            {
                Debug.LogError($"接收数据错误: {e.Message}");
                break;
            }
        }
    }
    
    private void ProcessData(string jsonData)
    {
        try
        {
            JObject data = JObject.Parse(jsonData);
            
            if (data.ContainsKey("attention"))
            {
                currentAttention = data["attention"].Value<float>();
                OnAttentionUpdated?.Invoke(currentAttention);
            }
            
            if (data.ContainsKey("imu"))
            {
                JObject imu = data["imu"].ToObject<JObject>();
                currentYaw = imu.ContainsKey("yaw") ? imu["yaw"].Value<float>() : 0f;
                currentPitch = imu.ContainsKey("pitch") ? imu["pitch"].Value<float>() : 0f;
                OnIMUUpdated?.Invoke(currentYaw, currentPitch);
            }
        }
        catch (Exception e)
        {
            Debug.LogError($"解析数据错误: {e.Message}");
        }
    }
    
    public float GetAttention() => currentAttention;
    public float GetYaw() => currentYaw;
    public float GetPitch() => currentPitch;
    public bool IsConnected() => isConnected;
}
```

// src/unity/Assets/Scripts/Player/PlayerController.cs

using UnityEngine;

public class PlayerController : MonoBehaviour
{
    [Header("移动设置")]
    public float baseSpeed = 10f;
    public float maxSpeed = 30f;
    public float minSpeed = 5f;
    public float rotationSpeed = 2f;
    public float verticalLimit = 10f;
    public float horizontalLimit = 20f;
    
    [Header("专注力映射")]
    public float attentionSmoothSpeed = 5f;
    
    private float currentSpeed;
    private float targetSpeed;
    private Vector3 startPosition;
    private BCIManager bciManager;
    
    void Start()
    {
        startPosition = transform.position;
        bciManager = BCIManager.Instance;
        
        if (bciManager != null)
        {
            bciManager.OnAttentionUpdated += OnAttentionUpdated;
            bciManager.OnIMUUpdated += OnIMUUpdated;
        }
    }
    
    void Update()
    {
        UpdateSpeed();
        UpdatePosition();
    }
    
    private void UpdateSpeed()
    {
        // 平滑过渡速度
        currentSpeed = Mathf.Lerp(currentSpeed, targetSpeed, Time.deltaTime * attentionSmoothSpeed);
        
        // 向前移动
        transform.Translate(Vector3.forward * currentSpeed * Time.deltaTime);
    }
    
    private void UpdatePosition()
    {
        if (bciManager == null) return;
        
        // 获取IMU数据
        float yaw = bciManager.GetYaw();
        float pitch = bciManager.GetPitch();
        
        // 计算移动
        float horizontalMove = yaw * rotationSpeed * Time.deltaTime;
        float verticalMove = pitch * rotationSpeed * Time.deltaTime;
        
        // 应用移动
        Vector3 newPosition = transform.position + new Vector3(horizontalMove, verticalMove, 0);
        
        // 限制范围
        newPosition.x = Mathf.Clamp(newPosition.x, -horizontalLimit, horizontalLimit);
        newPosition.y = Mathf.Clamp(newPosition.y, -verticalLimit, verticalLimit);
        
        transform.position = newPosition;
    }
    
    private void OnAttentionUpdated(float attention)
    {
        // 将专注力映射为速度
        // attention范围: 0-100
        // speed范围: minSpeed-maxSpeed
        targetSpeed = minSpeed + (maxSpeed - minSpeed) * (attention / 100f);
    }
    
    private void OnIMUUpdated(float yaw, float pitch)
    {
        // IMU数据已在Update中处理
    }
    
    void OnDestroy()
    {
        if (bciManager != null)
        {
            bciManager.OnAttentionUpdated -= OnAttentionUpdated;
            bciManager.OnIMUUpdated -= OnIMUUpdated;
        }
    }
    
    void OnCollisionEnter(Collision collision)
    {
        if (collision.gameObject.CompareTag("Obstacle"))
        {
            // 碰到障碍物
            GameManager.Instance.OnPlayerHitObstacle();
        }
    }
    
    void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Collectible"))
        {
            // 收集物品
            GameManager.Instance.OnCollectiblePickedUp();
            Destroy(other.gameObject);
        }
    }
}

// src/unity/Assets/Scripts/Core/GameManager.cs

using UnityEngine;
using UnityEngine.SceneManagement;

public class GameManager : MonoBehaviour
{
    public static GameManager Instance { get; private set; }
    
    [Header("游戏状态")]
    public GameState currentState = GameState.Menu;
    public int score = 0;
    public float gameTimer = 0f;
    public float sessionDuration = 300f; // 5分钟
    
    [Header("专注力统计")]
    public float averageAttention = 0f;
    public float maxAttention = 0f;
    public float focusStreak = 0f;
    public float maxFocusStreak = 0f;
    
    [Header("奖励设置")]
    public float focusThreshold = 70f;
    public float focusStreakThreshold = 10f;
    
    private float attentionSum = 0f;
    private int attentionCount = 0;
    private BCIManager bciManager;
    
    public enum GameState
    {
        Menu,
        Playing,
        Paused,
        GameOver
    }
    
    void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
        }
    }
    
    void Start()
    {
        bciManager = BCIManager.Instance;
        if (bciManager != null)
        {
            bciManager.OnAttentionUpdated += OnAttentionUpdated;
        }
    }
    
    void Update()
    {
        if (currentState == GameState.Playing)
        {
            UpdateGameTimer();
            UpdateFocusStreak();
        }
    }
    
    public void StartGame()
    {
        currentState = GameState.Playing;
        score = 0;
        gameTimer = 0f;
        attentionSum = 0f;
        attentionCount = 0;
        focusStreak = 0f;
        maxFocusStreak = 0f;
        
        SceneManager.LoadScene("StarGuardian");
    }
    
    public void PauseGame()
    {
        currentState = GameState.Paused;
        Time.timeScale = 0f;
    }
    
    public void ResumeGame()
    {
        currentState = GameState.Playing;
        Time.timeScale = 1f;
    }
    
    public void EndGame()
    {
        currentState = GameState.GameOver;
        CalculateStatistics();
        SaveSessionData();
    }
    
    private void UpdateGameTimer()
    {
        gameTimer += Time.deltaTime;
        
        if (gameTimer >= sessionDuration)
        {
            EndGame();
        }
    }
    
    private void UpdateFocusStreak()
    {
        if (bciManager == null) return;
        
        float attention = bciManager.GetAttention();
        
        if (attention >= focusThreshold)
        {
            focusStreak += Time.deltaTime;
            maxFocusStreak = Mathf.Max(maxFocusStreak, focusStreak);
        }
        else
        {
            focusStreak = 0f;
        }
    }
    
    private void OnAttentionUpdated(float attention)
    {
        attentionSum += attention;
        attentionCount++;
        averageAttention = attentionSum / attentionCount;
        maxAttention = Mathf.Max(maxAttention, attention);
    }
    
    public void OnPlayerHitObstacle()
    {
        // 减速惩罚
        score = Mathf.Max(0, score - 10);
    }
    
    public void OnCollectiblePickedUp()
    {
        // 加分
        score += 10;
        
        // 专注连击加分
        if (focusStreak >= focusStreakThreshold)
        {
            score += 20;
        }
    }
    
    private void CalculateStatistics()
    {
        // 计算最终统计
    }
    
    private void SaveSessionData()
    {
        // 保存会话数据到文件
        SessionData data = new SessionData
        {
            date = System.DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss"),
            duration = gameTimer,
            score = score,
            averageAttention = averageAttention,
            maxAttention = maxAttention,
            maxFocusStreak = maxFocusStreak
        };
        
        string json = JsonUtility.ToJson(data, true);
        string filename = $"session_{System.DateTime.Now:yyyyMMdd_HHmmss}.json";
        System.IO.File.WriteAllText(filename, json);
    }
    
    void OnDestroy()
    {
        if (bciManager != null)
        {
            bciManager.OnAttentionUpdated -= OnAttentionUpdated;
        }
    }
}

    public float averageAttention;

    public float maxFocusStreak;

# tests/test_integration.py

import unittest
import time
from src.python.bci_data_collector import BCIDataCollector
from src.python.signal_processor import SignalProcessor
from src.python.socket_server import SocketServer

class TestIntegration(unittest.TestCase):
    """集成测试"""
    
    def setUp(self):
        self.server = SocketServer()
        self.collector = BCIDataCollector()
        self.processor = SignalProcessor()
        
    def test_data_flow(self):
        """测试数据流通"""
        # 启动服务器
        self.server.start()
        time.sleep(1)
        
        # 连接客户端
        self.collector.start()
        time.sleep(1)
        
        # 发送测试数据
        test_data = {
            'timestamp': time.time(),
            'attention': 75.0,
            'imu': {'yaw': 10.0, 'pitch': 5.0}
        }
        self.server.send_data(test_data)
        time.sleep(0.1)
        
        # 验证数据接收
        attention = self.collector.get_attention()
        self.assertIsNotNone(attention)
        self.assertEqual(attention, 75.0)
        
        # 清理
        self.collector.stop()
        self.server.stop()
        
    def test_signal_processing(self):
        """测试信号处理"""
        # 测试专注力信号处理
        test_values = [50, 60, 70, 80, 90]
        results = []
        
        for value in test_values:
            processed = self.processor.process_attention(value)
            results.append(processed)
            
        # 验证平滑效果
        self.assertTrue(all(0 <= r <= 100 for r in results))
        
    def test_latency(self):
        """测试延迟"""
        # 启动服务器
        self.server.start()
        time.sleep(1)
        
        # 连接客户端
        self.collector.start()
        time.sleep(1)
        
        # 测量延迟
        start_time = time.time()
        test_data = {'timestamp': start_time, 'attention': 50.0}
        self.server.send_data(test_data)
        
        # 等待数据接收
        timeout = 1.0
        while time.time() - start_time < timeout:
            if self.collector.get_attention() is not None:
                break
            time.sleep(0.01)
            
        end_time = time.time()
        latency = (end_time - start_time) * 1000  # 转换为毫秒
        
        print(f"延迟: {latency:.2f}ms")
        self.assertLess(latency, 50)  # 延迟应小于50ms
        
        # 清理
        self.collector.stop()
        self.server.stop()

if __name__ == '__main__':
    unittest.main()
```

# tools/feedback_collector.py

import json
import pandas as pd
from datetime import datetime

class FeedbackCollector:
    """用户反馈收集器"""
    
    def __init__(self):
        self.feedbacks = []
        
    def add_feedback(self, user_id, feedback_type, content, rating):
        """添加反馈"""
        feedback = {
            'user_id': user_id,
            'timestamp': datetime.now().isoformat(),
            'type': feedback_type,
            'content': content,
            'rating': rating
        }
        self.feedbacks.append(feedback)
        
    def export_to_csv(self, filename):
        """导出为CSV"""
        df = pd.DataFrame(self.feedbacks)
        df.to_csv(filename, index=False, encoding='utf-8-sig')
        
    def generate_report(self):
        """生成报告"""
        df = pd.DataFrame(self.feedbacks)
        
        report = {
            'total_feedbacks': len(df),
            'average_rating': df['rating'].mean(),
            'feedback_by_type': df['type'].value_counts().to_dict(),
            'rating_distribution': df['rating'].value_counts().sort_index().to_dict()
        }
        
        return report
```
