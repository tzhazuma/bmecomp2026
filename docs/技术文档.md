```python
import numpy as np
from typing import Optional, List
from dataclasses import dataclass
from datetime import datetime

@dataclass
class EEGData:
    """EEG数据结构"""
    timestamp: float
    channel_1: float  # 主通道
    channel_2: float  # 参考通道1
    channel_3: float  # 参考通道2
    quality: float    # 信号质量

class EEGCollector:
    """EEG数据采集器"""
    
    def __init__(self, sample_rate: int = 256):
        self.sample_rate = sample_rate
        self.buffer: List[EEGData] = []
        self.max_buffer_size = sample_rate * 10  # 10秒缓冲
        
    def add_sample(self, data: EEGData):
        """添加采样数据"""
        self.buffer.append(data)
        if len(self.buffer) > self.max_buffer_size:
            self.buffer.pop(0)
            
    def get_latest_samples(self, n_samples: int) -> List[EEGData]:
        """获取最新N个采样"""
        return self.buffer[-n_samples:]
        
    def get_buffer(self) -> List[EEGData]:
        """获取完整缓冲区"""
        return self.buffer.copy()
        
    def clear_buffer(self):
        """清空缓冲区"""
        self.buffer.clear()
```

#### 2.2.1 滤波算法

```python
# src/python/filters.py

import numpy as np
from collections import deque
from typing import List

class MovingAverageFilter:
    """滑动平均滤波器"""
    
    def __init__(self, window_size: int = 10):
        self.window_size = window_size
        self.buffer = deque(maxlen=window_size)
        
    def process(self, value: float) -> float:
        """处理新数据"""
        self.buffer.append(value)
        return np.mean(list(self.buffer))
        
    def reset(self):
        """重置滤波器"""
        self.buffer.clear()


class ExponentialSmoothingFilter:
    """指数平滑滤波器"""
    
    def __init__(self, alpha: float = 0.7):
        self.alpha = alpha
        self.last_value = None
        
    def process(self, value: float) -> float:
        """处理新数据"""
        if self.last_value is None:
            self.last_value = value
        else:
            self.last_value = self.alpha * value + (1 - self.alpha) * self.last_value
        return self.last_value
        
    def reset(self):
        """重置滤波器"""
        self.last_value = None


class KalmanFilter:
    """卡尔曼滤波器"""
    
    def __init__(self, process_noise: float = 0.01, measurement_noise: float = 0.1):
        self.process_noise = process_noise
        self.measurement_noise = measurement_noise
        
        # 状态估计
        self.x = 0.0  # 状态
        self.P = 1.0  # 估计误差协方差
        
        # 状态转移矩阵
        self.F = 1.0
        
        # 观测矩阵
        self.H = 1.0
        
        # 过程噪声协方差
        self.Q = process_noise
        
        # 观测噪声协方差
        self.R = measurement_noise
        
    def process(self, measurement: float) -> float:
        """处理新测量值"""
        # 预测
        x_pred = self.F * self.x
        P_pred = self.F * self.P * self.F + self.Q
        
        # 更新
        K = P_pred * self.H / (self.H * P_pred * self.H + self.R)
        self.x = x_pred + K * (measurement - self.H * x_pred)
        self.P = (1 - K * self.H) * P_pred
        
        return self.x
        
    def reset(self):
        """重置滤波器"""
        self.x = 0.0
        self.P = 1.0
```

```python
# src/python/attention_calculator.py

import numpy as np
from typing import List, Optional
from dataclasses import dataclass

@dataclass
class AttentionResult:
    """专注力计算结果"""
    timestamp: float
    raw_value: float       # 原始值
    filtered_value: float  # 滤波后值
    normalized_value: float  # 归一化值 (0-100)
    quality: float         # 信号质量

class AttentionCalculator:
    """专注力计算器"""
    
    def __init__(self):
        self.min_value = 0.0
        self.max_value = 100.0
        self.calibration_data = []
        self.is_calibrated = False
        
    def calibrate(self, baseline_data: List[float]):
        """校准"""
        if baseline_data:
            self.min_value = np.percentile(baseline_data, 5)
            self.max_value = np.percentile(baseline_data, 95)
            self.is_calibrated = True
            
    def calculate(self, raw_eeg: float, alpha_power: float, beta_power: float) -> AttentionResult:
        """计算专注力"""
        # 专注力指标 = beta / (alpha + beta)
        if alpha_power + beta_power > 0:
            raw_attention = beta_power / (alpha_power + beta_power) * 100
        else:
            raw_attention = 0.0
            
        # 归一化
        if self.is_calibrated:
            normalized = np.clip(
                (raw_attention - self.min_value) / (self.max_value - self.min_value) * 100,
                0, 100
            )
        else:
            normalized = np.clip(raw_attention, 0, 100)
            
        return AttentionResult(
            timestamp=0.0,
            raw_value=raw_attention,
            filtered_value=0.0,
            normalized_value=normalized,
            quality=1.0
        )
```

```python
import json
from typing import Dict, Any
from dataclasses import dataclass, asdict

@dataclass
class BCIProtocol:
    """BCI通信协议"""
    
    @staticmethod
    def create_message(msg_type: str, data: Dict[str, Any]) -> str:
        """创建消息"""
        message = {
            'type': msg_type,
            'timestamp': time.time(),
            'data': data
        }
        return json.dumps(message)
        
    @staticmethod
    def parse_message(message: str) -> Dict[str, Any]:
        """解析消息"""
        try:
            return json.loads(message)
        except json.JSONDecodeError as e:
            raise ValueError(f"消息解析失败: {e}")
            
    @staticmethod
    def create_attention_message(attention: float, quality: float) -> str:
        """创建专注力消息"""
        return BCIProtocol.create_message('attention', {
            'value': attention,
            'quality': quality
        })
        
    @staticmethod
    def create_imu_message(yaw: float, pitch: float, roll: float) -> str:
        """创建IMU消息"""
        return BCIProtocol.create_message('imu', {
            'yaw': yaw,
            'pitch': pitch,
            'roll': roll
        })
        
    @staticmethod
    def create_status_message(status: str, battery: float) -> str:
        """创建状态消息"""
        return BCIProtocol.create_message('status', {
            'status': status,
            'battery': battery
        })
```

```csharp
// src/unity/Assets/Scripts/Network/SocketClient.cs

using UnityEngine;
using System;
using System.Net.Sockets;
using System.Text;
using System.Threading;
using Newtonsoft.Json.Linq;

public class SocketClient : MonoBehaviour
{
    [Header("连接设置")]
    public string serverHost = "127.0.0.1";
    public int serverPort = 5555;
    public float reconnectInterval = 5f;
    
    private TcpClient client;
    private NetworkStream stream;
    private Thread receiveThread;
    private bool isConnected = false;
    private bool shouldReconnect = true;
    
    public event Action<string> OnMessageReceived;
    public event Action OnConnected;
    public event Action OnDisconnected;
    
    void Start()
    {
        Connect();
    }
    
    void OnDestroy()
    {
        shouldReconnect = false;
        Disconnect();
    }
    
    public void Connect()
    {
        try
        {
            client = new TcpClient();
            client.Connect(serverHost, serverPort);
            stream = client.GetStream();
            isConnected = true;
            
            receiveThread = new Thread(ReceiveLoop);
            receiveThread.IsBackground = true;
            receiveThread.Start();
            
            OnConnected?.Invoke();
            Debug.Log("已连接到服务器");
        }
        catch (Exception e)
        {
            Debug.LogError($"连接失败: {e.Message}");
            if (shouldReconnect)
            {
                Invoke(nameof(Connect), reconnectInterval);
            }
        }
    }
    
    public void Disconnect()
    {
        isConnected = false;
        
        if (stream != null)
        {
            stream.Close();
            stream = null;
        }
        
        if (client != null)
        {
            client.Close();
            client = null;
        }
        
        OnDisconnected?.Invoke();
    }
    
    private void ReceiveLoop()
    {
        byte[] buffer = new byte[4096];
        
        while (isConnected)
        {
            try
            {
                if (stream.DataAvailable)
                {
                    int bytesRead = stream.Read(buffer, 0, buffer.Length);
                    if (bytesRead > 0)
                    {
                        string message = Encoding.UTF8.GetString(buffer, 0, bytesRead);
                        OnMessageReceived?.Invoke(message);
                    }
                }
                Thread.Sleep(10);
            }
            catch (Exception e)
            {
                Debug.LogError($"接收错误: {e.Message}");
                break;
            }
        }
        
        if (shouldReconnect)
        {
            Invoke(nameof(Connect), reconnectInterval);
        }
    }
    
    public void SendMessage(string message)
    {
        if (isConnected && stream != null)
        {
            try
            {
                byte[] data = Encoding.UTF8.GetBytes(message);
                stream.Write(data, 0, data.Length);
            }
            catch (Exception e)
            {
                Debug.LogError($"发送错误: {e.Message}");
            }
        }
    }
    
    public bool IsConnected => isConnected;
}
```

```csharp
        // 平滑过渡

            speedMapping.minAttention,

        // 映射到速度
        targetSpeed = Mathf.Lerp(
            speedMapping.minSpeed,
            speedMapping.maxSpeed,
            curvedAttention
        );
    }
    
    public float GetCurrentSpeed()

        return currentSpeed;
    }
}
```

```csharp
using UnityEngine;

[System.Serializable]
public class DirectionSettings
{
    public float yawSensitivity = 2f;
    public float pitchSensitivity = 1.5f;
    public float deadzone = 5f;
    public float maxAngle = 45f;
    public float smoothing = 0.1f;
}

public class DirectionController : MonoBehaviour
{
    [Header("方向控制设置")]
    public DirectionSettings settings;
    
    private float targetYaw;
    private float targetPitch;
    private float currentYaw;
    private float currentPitch;
    
    void Update()
    {
        // 平滑处理
        currentYaw = Mathf.Lerp(currentYaw, targetYaw, settings.smoothing);
        currentPitch = Mathf.Lerp(currentPitch, targetPitch, settings.smoothing);
        
        // 应用旋转
        ApplyRotation();
    }
    
    public void UpdateDirection(float yaw, float pitch)
    {
        // 死区处理
        if (Mathf.Abs(yaw) < settings.deadzone) yaw = 0f;
        if (Mathf.Abs(pitch) < settings.deadzone) pitch = 0f;
        
        // 应用灵敏度
        targetYaw = yaw * settings.yawSensitivity;
        targetPitch = pitch * settings.pitchSensitivity;
        
        // 限制角度
        targetYaw = Mathf.Clamp(targetYaw, -settings.maxAngle, settings.maxAngle);
        targetPitch = Mathf.Clamp(targetPitch, -settings.maxAngle, settings.maxAngle);
    }
    
    private void ApplyRotation()
    {
        // 应用偏航（左右转向）
        transform.Rotate(Vector3.up, currentYaw * Time.deltaTime);
        
        // 应用俯仰（上下倾斜）
        transform.Rotate(Vector3.right, currentPitch * Time.deltaTime);
    }
}
```

### 5.1 会话记录

```csharp
// src/unity/Assets/Scripts/Data/SessionRecorder.cs

using UnityEngine;
using System;
using System.Collections.Generic;
using System.IO;

[System.Serializable]
public class SessionRecord
{
    public string sessionId;
    public string startTime;
    public string endTime;
    public float duration;
    public int score;
    public float averageAttention;
    public float maxAttention;
    public float maxFocusStreak;
    public List<AttentionSample> attentionSamples;
}

[System.Serializable]
public class AttentionSample
{
    public float timestamp;
    public float value;
}

public class SessionRecorder : MonoBehaviour
{
    [Header("记录设置")]
    public bool recordEnabled = true;
    public float sampleInterval = 0.1f;  // 100ms
    
    private SessionRecord currentSession;
    private float lastSampleTime;
    
    void Start()
    {
        StartNewSession();
    }
    
    void Update()
    {
        if (recordEnabled && Time.time - lastSampleTime >= sampleInterval)
        {
            RecordSample();
            lastSampleTime = Time.time;
        }
    }
    
    void OnDestroy()
    {
        EndSession();
    }
    
    public void StartNewSession()
    {
        currentSession = new SessionRecord
        {
            sessionId = Guid.NewGuid().ToString(),
            startTime = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss"),
            attentionSamples = new List<AttentionSample>()
        };
    }
    
    public void RecordSample()
    {
        if (BCIManager.Instance != null)
        {
            float attention = BCIManager.Instance.GetAttention();
            
            currentSession.attentionSamples.Add(new AttentionSample
            {
                timestamp = Time.time,
                value = attention
            });
        }
    }
    
    public void EndSession()
    {
        if (currentSession != null)
        {
            currentSession.endTime = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");
            currentSession.duration = Time.time;
            
            CalculateStatistics();
            SaveSession();
        }
    }
    
    private void CalculateStatistics()
    {
        if (currentSession.attentionSamples.Count > 0)
        {
            float sum = 0;
            float max = 0;
            
            foreach (var sample in currentSession.attentionSamples)
            {
                sum += sample.value;
                max = Mathf.Max(max, sample.value);
            }
            
            currentSession.averageAttention = sum / currentSession.attentionSamples.Count;
            currentSession.maxAttention = max;
        }
    }
    
    private void SaveSession()
    {
        string json = JsonUtility.ToJson(currentSession, true);
        string filename = $"session_{currentSession.sessionId}.json";
        string path = Path.Combine(Application.persistentDataPath, filename);
        
        File.WriteAllText(path, json);
        Debug.Log($"会话已保存: {path}");
    }
}
```

```csharp
// src/unity/Assets/Scripts/Performance/PerformanceOptimizer.cs

using UnityEngine;

public class PerformanceOptimizer : MonoBehaviour
{
    [Header("性能设置")]
    public int targetFrameRate = 72;
    public bool enableVSync = false;
    public int antiAliasing = 4;
    
    [Header("LOD设置")]
    public float lodBias = 1.5f;
    public int maximumLODLevel = 0;
    
    void Start()
    {
        OptimizePerformance();
    }
    
    private void OptimizePerformance()
    {
        // 设置帧率
        Application.targetFrameRate = targetFrameRate;
        
        // 设置垂直同步
        QualitySettings.vSyncCount = enableVSync ? 1 : 0;
        
        // 设置抗锯齿
        QualitySettings.antiAliasing = antiAliasing;
        
        // 设置LOD
        QualitySettings.lodBias = lodBias;
        QualitySettings.maximumLODLevel = maximumLODLevel;
        
        // 设置阴影
        QualitySettings.shadows = ShadowQuality.HardOnly;
        QualitySettings.shadowResolution = ShadowResolution.Medium;
        
        // 设置渲染
        QualitySettings.pixelLightCount = 2;
        QualitySettings.anisotropicFiltering = AnisotropicFiltering.Enable;
    }
    
    void Update()
    {
        MonitorPerformance();
    }
    
    private void MonitorPerformance()
    {
        float fps = 1.0f / Time.deltaTime;
        
        if (fps < targetFrameRate * 0.9f)
        {
            Debug.LogWarning($"帧率过低: {fps:F1} FPS");
            ReduceQuality();
        }
    }
    
    private void ReduceQuality()
    {
        // 动态降低质量
        QualitySettings.antiAliasing = Mathf.Max(0, QualitySettings.antiAliasing / 2);
        QualitySettings.shadows = ShadowQuality.Disable;
    }
}
```
